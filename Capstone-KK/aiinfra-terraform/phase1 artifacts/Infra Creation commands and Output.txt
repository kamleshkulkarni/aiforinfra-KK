    # storage/02-disk-fill.sh

#!/bin/bash
FILL_TARGET=80
OUTPUT_DIR=/tmp/lab-disk-fill
mkdir -p \$OUTPUT_DIR
CUR_PCT=\$(df / | tail -1 | awk "{print \$5}" | tr -d "%")
echo "[DISK-FILL] Current disk usage: \${CUR_PCT}%"
if [ "\$CUR_PCT" -ge "\$FILL_TARGET" ]; then
  echo "[DISK-FILL] Already at \${CUR_PCT}% — no fill needed"
  exit 0
fi
TOTAL_KB=\$(df / | tail -1 | awk "{print \$2}")
USED_KB=\$(df  / | tail -1 | awk "{print \$3}")
TARGET_KB=\$(( TOTAL_KB * FILL_TARGET / 100 ))
WRITE_KB=\$(( TARGET_KB - USED_KB - 512000 ))
if [ "\$WRITE_KB" -le 0 ]; then
  echo "[DISK-FILL] Safety margin — skipping fill"
  exit 0
fi
WRITE_MB=\$(( WRITE_KB / 1024 ))
echo "[DISK-FILL] Writing \${WRITE_MB}MB to reach \${FILL_TARGET}%..."
dd if=/dev/zero of=\$OUTPUT_DIR/fill-file.dat bs=1M count=\$WRITE_MB status=progress
echo "[DISK-FILL] Disk now: \$(df / | tail -1 | awk "{print \$5}")"
echo "[DISK-FILL] Remove:   sudo rm -f \$OUTPUT_DIR/fill-file.dat"
ENDSCRIPT

# 01-cpu-stress.ps1
@'
param(
    [int]$DurationSeconds = 180,
    [int]$Workers = 2
)
Write-Host "[CPU-STRESS] Starting $Workers workers for $DurationSeconds seconds..."
Write-Host "[CPU-STRESS] Monitor: Get-Counter '\Processor(_Total)\% Processor Time' -SampleInterval 2 -MaxSamples 10"
$jobs = 1..$Workers | ForEach-Object {
    Start-Job -ScriptBlock {
        $end = (Get-Date).AddSeconds($using:DurationSeconds)
        while ((Get-Date) -lt $end) {
            [void][Math]::Sqrt([System.Random]::new().NextDouble() * 1000000)
        }
    }
}
Write-Host "[CPU-STRESS] $($jobs.Count) jobs running. IDs: $($jobs.Id -join ', ')"
Write-Host "[CPU-STRESS] Stop: Get-Job | Stop-Job; Get-Job | Remove-Job"

    '\Memory\Available MBytes', `
    '\Memory\% Committed Bytes In Use', `
    '\PhysicalDisk(_Total)\Avg. Disk sec/Read', `
    '\PhysicalDisk(_Total)\Avg. Disk sec/Write' `
